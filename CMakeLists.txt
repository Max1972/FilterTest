cmake_minimum_required(VERSION 3.22)
project(OpenMB VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OPENMB_BUILD_PLUGIN "Build JUCE plugin (requires platform development dependencies)" OFF)
option(OPENMB_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(openmb_dsp src/Dynamics.cpp)
target_include_directories(openmb_dsp PUBLIC include)
set_target_properties(openmb_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(openmb_dsp PRIVATE -Wall -Wextra -Wpedantic)
  if(OPENMB_SANITIZE)
    target_compile_options(openmb_dsp PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(openmb_dsp PUBLIC -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(openmb_tests tests/DynamicsTests.cpp)
  target_link_libraries(openmb_tests PRIVATE openmb_dsp)
  add_test(NAME dynamics COMMAND openmb_tests)
endif()
if(OPENMB_BUILD_PLUGIN)
  if(APPLE)
    if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
      set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum macOS version" FORCE)
    endif()
  endif()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 7.0.12 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
  set(formats Standalone)
  if(APPLE)
    list(APPEND formats AU)
  endif()
  juce_add_plugin(OpenMB COMPANY_NAME "OpenMB" PLUGIN_MANUFACTURER_CODE Opmb
    PLUGIN_CODE Omb1 FORMATS ${formats} PRODUCT_NAME "OpenMB" IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(OpenMB PRIVATE plugin/Processor.cpp)
  target_link_libraries(OpenMB PRIVATE openmb_dsp juce::juce_audio_utils)
  target_compile_definitions(OpenMB PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
endif()
